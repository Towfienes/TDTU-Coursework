\chapter{TRAINING MODELS FROM SCRATCH}
\section{Linear regression}
For a training set with $n$ examples and $d$ features, let $X\in\mathbb{R}^{n\times d}$, $w\in\mathbb{R}^{d}$ and $b\in\mathbb{R}$. The predictions are $\hat y=Xw+b\mathbf{1}$ and the mean half-squared-error loss is
\[
J(w,b)=\frac{1}{2n}\sum_{i=1}^{n}(w^{\mathsf T}x_i+b-y_i)^2.
\]
Differentiating gives the following simultaneous gradient descent updates:
\[
\begin{aligned}
w^{(t+1)}&=w^{(t)}-\eta\frac{1}{n}X^{\mathsf T}(Xw^{(t)}+b^{(t)}\mathbf1-y),\\
b^{(t+1)}&=b^{(t)}-\eta\frac{1}{n}\mathbf1^{\mathsf T}(Xw^{(t)}+b^{(t)}\mathbf1-y).
\end{aligned}
\]
